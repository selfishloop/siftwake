+struct Commit {
+    arrival: Arrival,
+    start: Instant,
+    end: Instant,
+    id: usize,
+    body_bytes: usize,
+    old_revision: u64,
+    sequence: u64,
+}
+
+fn see(view: &impl View, check: &Check) -> Result<Instant> {
+    let marker = churn::marker(check.id, check.revision);
+    let page = view.first(&query(&marker))?;
+    let first = Instant::now();
+    assert_eq!(page.validation, ValidationStatus::Verified);
+    assert_eq!(page.sequence, view.sequence());
+    assert!(page.sequence >= check.sequence);
+    assert_eq!(page.total, MatchTotal::Exact(1));
+    assert_eq!(page.hits.len(), 1);
+    assert_eq!(page.hits[0].document.key, churn::key(check.id));
+    assert_eq!(page.hits[0].version, check.sequence);
+    Ok(first)
+}
+
+fn reader<F: Factory>(
+    factory: F,
+    queries: Vec<Query>,
+    checks: Receiver<Check>,
+    observations: Sender<Observation>,
+    stop: &AtomicBool,
+) -> Result<()> {
+    let mut client = factory.open()?;
+    let mut index = 0;
+    while !stop.load(Ordering::Acquire) {
+        match checks.try_recv() {
+            Ok(check) => {
+                let begin = Instant::now();
+                let (view, _) = client.acquire()?;
+                let acquired = Instant::now();
+                let first = see(&view, &check)?;
+                let sequence = view.sequence();
+                assert!(view.release()?.cleanup_error.is_none());
+                observations.send(Observation {
+                    begin,
+                    acquired,
+                    first,
+                    sequence,
+                })?;
+            }
+            Err(TryRecvError::Empty) => {
+                let (view, _) = client.acquire()?;
+                let page = view.first(&queries[index])?;
+                assert_eq!(page.sequence, view.sequence());
+                assert_eq!(page.validation, ValidationStatus::Verified);
+                assert!(view.release()?.cleanup_error.is_none());
+                index = (index + 1) % queries.len();
+            }
+            Err(TryRecvError::Disconnected) => return Ok(()),
+        }
+    }
+    Ok(())
+}
+
+struct ClientQueue {
+    checks: Sender<Check>,
+    observations: Receiver<Observation>,
+}
+
+fn ms(duration: Duration) -> f64 {
+    duration.as_secs_f64() * 1000.
+}
+
+fn paired_ids(corpus: &Corpus, model: &Model, cycle: usize) -> [usize; 4] {
+    BANDS.map(|(low, high)| {
+        let eligible: Vec<_> = model
+            .rows
+            .iter()
+            .filter(|(_, row)| {
+                let n = corpus.index.entries[row.body].text_bytes;
+                low <= n && n < high
+            })
+            .map(|(&id, _)| id)
+            .collect();
+        eligible[(cycle * 701 + 17) % eligible.len()]
+    })
+}
+
+fn commit_event(
+    db: &mut impl Backend,
+    corpus: &mut Corpus,
+    model: &mut Model,
+    id: usize,
+    arrival: Arrival,
+    queue: &ClientQueue,
+) -> Result<Commit> {
+    let old = model.rows[&id].clone();
+    let new = Row {
+        body: old.body,
+        revision: old.revision + 1,
+        version: 0,
+    };
+    let change = Change {
+        id,
+        old: Some(old.clone()),
+        new: Some(new.clone()),
+        kind: "update".into(),
+    };
+    let start = Instant::now();
+    let sequence = db.commit(corpus, std::slice::from_ref(&change), true)?;
+    let end = Instant::now();
+    churn::apply(model, std::slice::from_ref(&change), sequence);
+    queue.checks.send(Check {
+        id,
+        revision: new.revision,
+        sequence,
+    })?;
+    Ok(Commit {
+        arrival,
+        start,
+        end,
+        id,
+        body_bytes: corpus.index.entries[new.body].text_bytes,
+        old_revision: old.revision,
+        sequence,
+    })
+}
+
+fn producer(
+    origin: Instant,
+    sender: Sender<Arrival>,
+    phase: (
+        std::sync::Arc<std::sync::atomic::AtomicU8>,
+        std::sync::Arc<std::sync::atomic::AtomicU8>,
+    ),
+) -> Result<()> {
+    for (index, offset) in OFFSETS_MS.into_iter().enumerate() {
+        let deadline = origin + Duration::from_millis(offset);
+        while let Some(remaining) = deadline.checked_duration_since(Instant::now()) {
+            std::thread::sleep(remaining);
+        }
+        sender.send(Arrival {
+            index,
+            at: Instant::now(),
+            worker_phase: phase.0.load(Ordering::Acquire),
+            writer_phase: phase.1.load(Ordering::Acquire),
+        })?;
+    }
+    Ok(())
+}
+
+fn measure_phase(
+    db: &mut impl Backend,
+    corpus: &mut Corpus,
+    model: &mut Model,
+    queue: &ClientQueue,
+    ids: [usize; 4],
+    maintenance: bool,
+) -> Result<Vec<Value>> {
+    let (send, receive) = mpsc::channel();
+    let phase = db.arrival_phases();
+    std::thread::scope(|scope| {
+        let origin = Instant::now();
+        let worker = scope.spawn(move || producer(origin, send, phase));
+        let mut commits = Vec::new();
+        for _ in ids {
+            let arrival = if maintenance {
+                next_with_maintenance(db, &receive)?
+            } else {
+                receive.recv_timeout(TIMEOUT)?
+            };
+            let id = ids[arrival.index];
+            commits.push(commit_event(db, corpus, model, id, arrival, queue)?);
+        }
+        worker.join().map_err(|_| "arrival producer panicked")??;
+        let mut output = Vec::new();
+        for c in &commits {
+            let observed = match queue.observations.recv_timeout(TIMEOUT) {
+                Err(RecvTimeoutError::Timeout) => {
+                    return Err("visibility response timed out".into());
+                }
+                other => other?,
+            };
+            assert!(observed.begin >= c.end);
+            assert!(observed.sequence >= c.sequence);
+            output.push(json!({
+                "condition": if maintenance { "background" } else { "control" },
+                "worker_phase": c.arrival.worker_phase, "writer_phase": c.arrival.writer_phase,
+                "event_index": c.arrival.index, "offset_ms": OFFSETS_MS[c.arrival.index],
+                "arrival_ms": ms(c.arrival.at.duration_since(origin)),
+                "id": c.id, "body_bytes": c.body_bytes, "sequence": c.sequence, "snapshot_sequence": observed.sequence,
+                "queue_ms": ms(c.start.duration_since(c.arrival.at)),
+                "commit_ms": ms(c.end.duration_since(c.start)),
+                "commit_to_reader_ms": ms(observed.begin.duration_since(c.end)),
+                "acquire_ms": ms(observed.acquired.duration_since(observed.begin)),
+                "first_query_ms": ms(observed.first.duration_since(observed.acquired)),
+                "arrival_to_visible_ms": ms(observed.first.duration_since(c.arrival.at)),
+                "oracle": true
+            }));
+        }
+        // These exact GET and stale-marker checks are deliberately outside the
+        // observation loop so that one event cannot slow down the next probe.
+        for c in &commits {
+            let row = &model.rows[&c.id];
+            assert_eq!(
+                db.get(c.id)?,
+                Some(fdb::StoredDocument {
+                    document: corpus.document(c.id, row)?,
+                    version: c.sequence,
+                })
+            );
+            assert_eq!(
+                db.search(&churn::marker(c.id, c.old_revision), 0, 1)?.total,
+                0
+            );
+        }
+        Ok(output)
+    })
+}
+
+fn next_with_maintenance(db: &mut impl Backend, receive: &Receiver<Arrival>) -> Result<Arrival> {
+    let stop = Instant::now() + TIMEOUT;
+    while Instant::now() < stop {
+        match receive.try_recv() {
+            Ok(arrival) => return Ok(arrival),
+            Err(TryRecvError::Empty) => {}
+            Err(error) => return Err(error.into()),
+        }
+        db.poll_idle()?;
+        match receive.recv_timeout(Duration::from_millis(5)) {
+            Ok(arrival) => return Ok(arrival),
+            Err(RecvTimeoutError::Timeout) => {}
+            Err(error) => return Err(error.into()),
+        }
+    }
+    Err("event queue timeout".into())
+}
+
+fn workload(
+    db: &mut impl Backend,
+    corpus: &mut Corpus,
+    model: &mut Model,
+    state: &str,
+    path: &str,
+    queue: &ClientQueue,
+) -> Result<()> {
+    for cycle in 0..CYCLES {
+        for group in changes(model, 30 + cycle).as_chunks::<100>().0 {
+            let sequence = db.commit(corpus, group, true)?;
+            churn::apply(model, group, sequence);
+        }
+        model.save(state)?;
+        let ids = paired_ids(corpus, model, cycle);
+        let order = if cycle % 2 == 0 {
+            [true, false]
+        } else {
+            [false, true]
+        };
+        for maintenance in order {
+            if !maintenance {
+                db.arrival_reclaim(false)?;
+            }
+            let events = measure_phase(db, corpus, model, queue, ids, maintenance)?;
+            model.save(state)?;
+            churn::emit(json!({"phase":"arrival_events","cycle":cycle+1,"events":events}));
+            if !maintenance {
+                db.arrival_reclaim(true)?;
+            }
+        }
+        churn::emit(
+            json!({"phase":"arrival_cycle","cycle":cycle+1,"stats":db.stats()?,"disk":churn::disk(path)?,"revision_installs":db.arrival_installs()}),
+        );
+    }
+    Ok(())
+}
+
+pub(super) fn run(
+    db: &mut impl Writer,
+    corpus: &mut Corpus,
+    model: &mut Model,
+    state: &str,
+    path: &str,
+) -> Result<()> {
+    assert_eq!(model.rows.len(), 50000);
+    let factory = db.factory()?;
+    let queries = corpus
+        .index
+        .cases
+        .iter()
+        .map(|case| query(&case.text))
+        .collect();
+    let (check_tx, check_rx) = mpsc::channel();
+    let (seen_tx, seen_rx) = mpsc::channel();
+    let stop = AtomicBool::new(false);
+    std::thread::scope(|scope| -> Result<()> {
+        let task = scope.spawn(|| reader(factory, queries, check_rx, seen_tx, &stop));
+        let guard = StopReader(&stop);
+        let queue = ClientQueue {
+            checks: check_tx,
+            observations: seen_rx,
+        };
+        let result = workload(db, corpus, model, state, path, &queue);
+        drop(guard);
+        task.join().map_err(|_| "snapshot reader panicked")??;
+        result
+    })?;
+    Ok(())
+}
